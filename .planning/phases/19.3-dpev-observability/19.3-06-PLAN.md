    ```typescript
    // Extract execution step events for replay (D-17 parity: include command/risk/target/stdout/stderr)
    if (
      eventType === 'execution_start' ||
      eventType === 'step_complete' ||
      eventType === 'step_failed'
    ) {
      const meta = entry.metadata as Record<string, unknown> | undefined;
      if (!meta) continue;
      const stepIndex = (meta.stepIndex as number) ?? executionSteps.length;
      const existingIdx = executionSteps.findIndex(s => s.stepIndex === stepIndex);

      const baseFields: Partial<StepState> = {
        command: (meta.command as string) ?? undefined,
        risk: (meta.risk as string) ?? undefined,
        target: (meta.target as string) ?? undefined,
        total: (meta.totalSteps as number) ?? undefined,
        stdout: (meta.stdout as string) ?? undefined,
        stderr: (meta.stderr as string) ?? undefined,
      };

      let status: StepState['status'];
      if (eventType === 'execution_start') status = 'pending';
      else if (eventType === 'step_failed') status = 'failed';
      else status = 'success';

      if (existingIdx >= 0) {
        // Merge: prefer new status + stdout/stderr; keep prior command/risk/target
        const prior = executionSteps[existingIdx];
        executionSteps[existingIdx] = {
          ...prior,
          status,
          stdout: baseFields.stdout ?? prior.stdout,
          stderr: baseFields.stderr ?? prior.stderr,
          command: baseFields.command ?? prior.command,
          risk: baseFields.risk ?? prior.risk,
          target: baseFields.target ?? prior.target,
          total: baseFields.total ?? prior.total,
        };
      } else {
        executionSteps.push({
          stepIndex,
          total: baseFields.total ?? 0,
          command: baseFields.command ?? '',
          risk: baseFields.risk ?? '',
          status,
          stdout: baseFields.stdout,
          stderr: baseFields.stderr,
          target: baseFields.target,
        });
      }
    }
    ```
